local function updateESP(plr)
    local data=espObjects[plr]
    if not data or not plr.Character then return end
    local char,root,hum=plr.Character,plr.Character:FindFirstChild("HumanoidRootPart"),plr.Character:FindFirstChild("Humanoid")
    for _,l in ipairs(data.Box) do l.Visible=false end
    data.HP_BG.Visible=false;data.HP_Fill.Visible=false
    for _,l in pairs(data.Skeleton) do l.Visible=false end
    data.NameTag.Visible=false;data.DistTag.Visible=false;data.Tracer.Visible=false
    if not root or not hum or hum.Health<=0 then return end
    local isEnemy=true
    if Settings.ESPTeamCheck and plr.Team==LocalPlayer.Team and LocalPlayer.Team~=nil then isEnemy=false end
    if not Settings.ESPEnabled or not isEnemy then return end
    local pos,onScreen=Camera:WorldToViewportPoint(root.Position)
    if not onScreen then return end
    local headPos=char:FindFirstChild("Head")
    local headScreen,headOn=Camera:WorldToViewportPoint(headPos and headPos.Position or root.Position+Vector3.new(0,3,0))
    local feetScreen,feetOn=Camera:WorldToViewportPoint(root.Position-Vector3.new(0,3,0))
    if not headOn or not feetOn then return end
    local height=math.abs(feetScreen.Y-headScreen.Y)
    local left,right=pos.X-(height*0.6)/2,pos.X+(height*0.6)/2
    if Settings.ESPBox then
        data.Box[1].From=Vector2.new(left,headScreen.Y);data.Box[1].To=Vector2.new(right,headScreen.Y)
        data.Box[2].From=Vector2.new(left,feetScreen.Y);data.Box[2].To=Vector2.new(right,feetScreen.Y)
        data.Box[3].From=Vector2.new(left,headScreen.Y);data.Box[3].To=Vector2.new(left,feetScreen.Y)
        data.Box[4].From=Vector2.new(right,headScreen.Y);data.Box[4].To=Vector2.new(right,feetScreen.Y)
        for _,l in ipairs(data.Box) do l.Visible=true;l.Thickness=Settings.ESPThickness;l.Color=Settings.ESPColor end
    end
    if Settings.ESPSkeleton then
        local j={Head=char:FindFirstChild("Head"),UpperTorso=char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso"),LowerTorso=char:FindFirstChild("LowerTorso") or char:FindFirstChild("Torso"),LeftUpperArm=char:FindFirstChild("LeftUpperArm"),LeftLowerArm=char:FindFirstChild("LeftLowerArm"),RightUpperArm=char:FindFirstChild("RightUpperArm"),RightLowerArm=char:FindFirstChild("RightLowerArm"),LeftUpperLeg=char:FindFirstChild("LeftUpperLeg"),LeftLowerLeg=char:FindFirstChild("LeftLowerLeg"),RightUpperLeg=char:FindFirstChild("RightUpperLeg"),RightLowerLeg=char:FindFirstChild("RightLowerLeg")}
        for pn,l in pairs(data.Skeleton) do
            local a,b=pn:match("([^_]+)_(.+)")
            local p1,p2=j[a],j[b]
            if p1 and p2 then
                local o1,on1=Camera:WorldToViewportPoint(p1.Position)
                local o2,on2=Camera:WorldToViewportPoint(p2.Position)
                if on1 and on2 then l.From=Vector2.new(o1.X,o1.Y);l.To=Vector2.new(o2.X,o2.Y);l.Visible=true;l.Thickness=Settings.ESPThickness;l.Color=Settings.ESPChamsColor end
            end
        end
    end
    if Settings.ESPHealthBar then
        local hp=math.clamp(hum.Health/hum.MaxHealth,0,1)
        local hh,hx=height*hp,left-6
        data.HP_BG.From=Vector2.new(hx,headScreen.Y);data.HP_BG.To=Vector2.new(hx,feetScreen.Y);data.HP_BG.Visible=true
        data.HP_Fill.From=Vector2.new(hx,feetScreen.Y);data.HP_Fill.To=Vector2.new(hx,feetScreen.Y-hh)
        data.HP_Fill.Color=Color3.fromRGB(255*(1-hp),255*hp,0);data.HP_Fill.Visible=true
    end
    if Settings.ESPName then data.NameTag.Text=plr.Name;data.NameTag.Position=Vector2.new(pos.X,headScreen.Y-15);data.NameTag.Visible=true end
    if Settings.ESPDistance then data.DistTag.Text=math.floor((root.Position-Camera.CFrame.Position).Magnitude).."m";data.DistTag.Position=Vector2.new(pos.X,feetScreen.Y+5);data.DistTag.Visible=true end
    if Settings.ESPTracers then data.Tracer.From=Vector2.new(Camera.ViewportSize.X/2,Camera.ViewportSize.Y);data.Tracer.To=Vector2.new(pos.X,pos.Y);data.Tracer.Visible=true;data.Tracer.Color=Settings.ESPTracersColor end

local function isPlayerVisible(tc)
    local o,tp=Camera.CFrame.Position,tc:FindFirstChild(Settings.AimTargetPart) or tc:FindFirstChild("Head")
    if not tp then return false end
    local pr=RaycastParams.new();pr.FilterType=Enum.RaycastFilterType.Exclude
    if LocalPlayer.Character then pr.FilterDescendantsInstances={LocalPlayer.Character} end
    local r=Workspace:Raycast(o,(tp.Position-o)*1000,pr)
    return r and r.Instance:IsDescendantOf(tc) or false
end
local function getBestTarget(fov,checkVis)
    local bt,sd,sc=nil,fov or Settings.AimFOV,Vector2.new(Camera.ViewportSize.X/2,Camera.ViewportSize.Y/2)
    for _,p in pairs(Players:GetPlayers()) do
        if p~=LocalPlayer and p.Character then
            if Settings.AimTeamCheck and p.Team==LocalPlayer.Team and LocalPlayer.Team~=nil then continue end
            local h,tp=p.Character:FindFirstChild("Humanoid"),p.Character:FindFirstChild(Settings.AimTargetPart) or p.Character:FindFirstChild("Head")
            if h and h.Health>0 and tp then
                local sp,os=Camera:WorldToViewportPoint(tp.Position)
                if os then
                    local d=(Vector2.new(sp.X,sp.Y)-sc).Magnitude
                    if d<sd then
                        if not checkVis or isPlayerVisible(p.Character) then sd=d;bt=p end
                    end
                end
            end
        end
    end
    return bt
end
-- Fly vars
local flyBV,flyBG=nil,nil
-- NoClip (без проваливания)
local function applyNoClip()
    pcall(function()
        local ch=LocalPlayer.Character
        if not ch then return end
        local root=ch:FindFirstChild("HumanoidRootPart")
        if root then
            root.CanCollide=false
            for _,p in pairs(ch:GetChildren()) do if p:IsA("BasePart") and p~=root then p.CanCollide=false end end
            if root.Position.Y<-100 then root.CFrame=CFrame.new(root.Position.X,10,root.Position.Z) end
        end
    end)
end
-- Fly
local function applyFly()
    pcall(function()
        local ch=LocalPlayer.Character
        if not ch then return end
        if Settings.Fly then
            local root=ch:FindFirstChild("HumanoidRootPart")
            if root then
                if not flyBV then flyBV=Instance.new("BodyVelocity");flyBV.MaxForce=Vector3.new(9e9,9e9,9e9);flyBV.Velocity=Vector3.new(0,0,0);flyBV.Parent=root end
                if not flyBG then flyBG=Instance.new("BodyGyro");flyBG.MaxTorque=Vector3.new(9e9,9e9,9e9);flyBG.P=9e4;flyBG.Parent=root end
                local md=Vector3.new(0,0,0)
                if UserInputService:IsKeyDown(Enum.KeyCode.W) then md=md+Camera.CFrame.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.S) then md=md-Camera.CFrame.LookVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.A) then md=md-Camera.CFrame.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.D) then md=md+Camera.CFrame.RightVector end
                if UserInputService:IsKeyDown(Enum.KeyCode.Space) then md=md+Vector3.new(0,1,0) end
                if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then md=md-Vector3.new(0,1,0) end
                flyBV.Velocity=md.Magnitude>0 and md.Unit*Settings.FlySpeed or Vector3.new(0,0,0)
                flyBG.CFrame=Camera.CFrame
            end
        else
            if flyBV then flyBV:Destroy();flyBV=nil end
            if flyBG then flyBG:Destroy();flyBG=nil end
        end
    end)
end

-- Rapid Fire через эмуляцию кликов
local lastShot=0
local function applyRapidFire()
    if Settings.RapidFire and UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) then
        local now=tick()
        if now-lastShot>=Settings.RapidFireDelay then
            lastShot=now
            pcall(function() mouse1click() end)
        end
    end
end
-- BHop

    pcall(function()
        local ch=LocalPlayer.Character
        if not ch then return end
        local hum=ch:FindFirstChild("Humanoid")
        if hum and Settings.BHop and UserInputService:IsKeyDown(Enum.KeyCode.Space) then

        end
    end)
end
-- Anti Recoil: компенсация дёргов камеры вверх при стрельбе
local lastLookY=nil
local function applyAntiRecoil(shooting)
    pcall(function()
        local curY=Camera.CFrame.LookVector.Y
        if shooting and lastLookY then
            local delta=curY-lastLookY
            if delta>0.001 then Camera.CFrame=Camera.CFrame*CFrame.Angles(-delta*0.9,0,0) end
        end
        lastLookY=Camera.CFrame.LookVector.Y
    end)
end

RunService.RenderStepped:Connect(function()
    if fovCircle then fovCircle.Visible=Settings.AimShowFOV and Settings.AimEnabled;fovCircle.Radius=Settings.AimFOV;fovCircle.Position=Vector2.new(Camera.ViewportSize.X/2,Camera.ViewportSize.Y/2) end

    local shooting=UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1)
    if Settings.AimEnabled then
        local tg=getBestTarget(nil,Settings.AimVisibleCheck)
        if tg and tg.Character then
            local tp=tg.Character:FindFirstChild(Settings.AimTargetPart) or tg.Character:FindFirstChild("Head")
            if tp then Camera.CFrame=Camera.CFrame:Lerp(CFrame.new(Camera.CFrame.Position,tp.Position),math.clamp(1-(Settings.AimSmooth/110),0.01,0.99)) end
        end
    end
    -- Bullet Manipulation: пока стреляешь камера в голову = пули в голову
    if (Settings.BulletManipulation or Settings.SilentAim) and shooting then
        local tg=getBestTarget(Settings.SilentAimFOV,true)
        if tg and tg.Character then
            local tp=tg.Character:FindFirstChild(Settings.AimTargetPart) or tg.Character:FindFirstChild("Head")
            if tp then Camera.CFrame=CFrame.new(Camera.CFrame.Position,tp.Position) end
        end
    end
    -- Anti Spread: стабилизация прицела при стрельбе
    if Settings.AntiSpread and shooting then
